// 生日倒计时计算 function updateCountdown() { // 设置生日日期：2025年11月8日 const birthday = new Date('2025-11-08'); const now = new Date(); // 如果生日已经过去，计算下一个生日 if (now > birthday) { birthday.setFullYear(birthday.getFullYear() + 1); } const diff = birthday - now; const days = Math.floor(diff / (1000 * 60 * 60 * 24)); const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60)); const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)); const seconds = Math.floor((diff % (1000 * 60)) / 1000); document.getElementById('countdown').textContent = `距离你的生日还有：${days}天 ${hours}小时 ${minutes}分钟 ${seconds}秒`; }
